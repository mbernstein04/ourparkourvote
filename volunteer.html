// ADD THIS SCRIPT BLOCK TO index.html just before the closing tag // It intercepts the hero signup form, posts to Netlify, then redirects // to /thank-you with the contact info in the URL so the shift form can use it. document.querySelector('form[name="hero-signup"]').addEventListener('submit', function(e) { e.preventDefault(); var form = this; var data = new FormData(form); fetch('/', { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: new URLSearchParams(data).toString() }) .then(function() { var params = new URLSearchParams({ 'first-name': data.get('first-name') || '', 'last-name': data.get('last-name') || '', 'email': data.get('email') || '', 'phone': data.get('phone') || '' }); window.location.href = '/thank-you?' + params.toString(); }) .catch(function() { // Fallback: just submit normally form.setAttribute('action', '/thank-you'); form.submit(); }); });
